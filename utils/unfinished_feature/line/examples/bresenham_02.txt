	line.col = table.col;

#include"line.h"

int	main(int len, char **str)
{
	t_table_fdf	table;
	int			output;
	t_line		line;
	t_line		boundary;
	t_ink		ink;

	ink.color = 75;
	ink.channel = HEIGHT;
	ink.thickness = 1;
	if (len < 3)
		return (0);
	output = open_dir_file(str[2], NULL, APPEND);
	table = init_table_fdf(30, 30, false);
	// table = open_table_fdf_file(str[1], NULL, parse_ascii_line_cheche01, false);
	boundary.p1.x = 0;
	boundary.p1.y = 0;
	boundary.p2.x = table.col - 1;
	boundary.p2.y = table.row - 1;
	line.p1.x = 2;
	line.p1.y = 3;
	line.p2.x = 10;
	line.p2.y = 15;

	write_table_ascii_cheche01(output, &table, HEIGHT);
	free_table_fdf(&table);
	return (0);
}
