#include "aura.h"

int	aura_complex_pythagorus(int x, int y, int z,
	t_float2d (*complex_func)(t_float2d a))
{
	t_float2d	complex;

	complex.x = (float)x;
	complex.y = (float)y;
	if (complex_func != NULL)
		complex = complex_func(complex);
	return (aura_pythagoras((int)complex.x, (int)complex.y, z));
}
